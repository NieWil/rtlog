cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(rtlog_benchmark)
option(RTLOG_BENCH_FOOTPRINT "Build the matched ten-site footprint workload" OFF)
option(RTLOG_BENCH_STRESS "Build the matched overload and interrupt workload" OFF)
if(RTLOG_BENCH_STRESS)
	target_sources(app PRIVATE src/stress.c)
elseif(RTLOG_BENCH_FOOTPRINT)
	target_sources(app PRIVATE src/footprint.c)
elseif(CONFIG_LOG)
	target_sources(app PRIVATE src/zephyr_log.c)
else()
	target_sources(app PRIVATE src/main.c)
endif()
